Replit Prompt — AiPM Change Order (COR) Module

Goal: Create a FastAPI app that adds a Change Order (COR) module tied to a project database and forecast reporting. It must:

Track COR originator (RFI, Addendum, Email, etc.) and require originator backup.

Require written approval backup before a COR can convert to a PO (hard gate).

Apply approved CORs to either existing cost codes (budget adjustments) or create new scope lines.

Expose reporting endpoints for pending/approved COR values and aging, so the Contract Forecast Report can consume them.

Store uploads under /mnt/data/uploads/<project_id>/<cor_id>/….

Data Model (minimum viable)

Project: id, number (unique), name

CostCode: id, project_id, code, description

ChangeOrder (COR):

id, project_id, cor_number (e.g., RIV-001), title, description

type (BUDGET_ADJUSTMENT | ADDED_SCOPE)

originator (RFI | ADDENDUM | EMAIL | OWNER_DIRECTIVE | OTHER)

status (DRAFT | SUBMITTED | APPROVED | REJECTED | PO_CREATED)

estimated_cost, final_cost

existing_costcode_id (nullable; if null and type is ADDED_SCOPE, requires new scope fields)

new_scope_code, new_scope_description (nullable unless ADDED_SCOPE)

created_at, approved_at, po_created_at

ChangeOrderDoc: id, cor_id, kind (ORIGINATOR_BACKUP | APPROVAL_BACKUP | OTHER), filename, path

PurchaseOrder (minimal): id, project_id, po_number, cor_id (unique), amount, costcode_id

Rules (gatekeeping)

Cannot SUBMIT unless at least one ORIGINATOR_BACKUP is attached.

Cannot APPROVE unless at least one APPROVAL_BACKUP is attached.

Cannot CREATE PO unless COR is APPROVED and final_cost is set.

If type = BUDGET_ADJUSTMENT then existing_costcode_id is required.

If type = ADDED_SCOPE, then new_scope_code and new_scope_description are required; PO creation will auto-provision a new CostCode if it doesn’t exist yet.

Reporting

Expose JSON for:

/api/reporting/cor/summary?project=<project_number> → pending total, approved total, count by status, % growth.

/api/reporting/cor/aging?project=<project_number> → days in status (draft→submitted→approved→po).

These endpoints will be pulled into your Contract Forecast Report builder to combine with base contract/budget.

def get_db():
    from contextlib import contextmanager
    @contextmanager
    def _session():
        db = SessionLocal()
        try:
            yield db
            db.commit()
        except:
            db.rollback()
            raise
        finally:
            db.close()
    return _session()

from sqlalchemy import String, Enum, ForeignKey, DateTime, Numeric, UniqueConstraint
from sqlalchemy.orm import Mapped, mapped_column, relationship
from datetime import datetime
import enum
from db import Base

class CorStatus(enum.Enum):
    DRAFT = "DRAFT"
    SUBMITTED = "SUBMITTED"
    APPROVED = "APPROVED"
    REJECTED = "REJECTED"
    PO_CREATED = "PO_CREATED"

class CorType(enum.Enum):
    BUDGET_ADJUSTMENT = "BUDGET_ADJUSTMENT"
    ADDED_SCOPE = "ADDED_SCOPE"

class DocKind(enum.Enum):
    ORIGINATOR_BACKUP = "ORIGINATOR_BACKUP"
    APPROVAL_BACKUP = "APPROVAL_BACKUP"
    OTHER = "OTHER"

class Project(Base):
    __tablename__ = "projects"
    id: Mapped[int] = mapped_column(primary_key=True)
    number: Mapped[str] = mapped_column(String(50), unique=True, index=True)
    name: Mapped[str] = mapped_column(String(200))
    costcodes: Mapped[list["CostCode"]] = relationship(back_populates="project")
    cors: Mapped[list["ChangeOrder"]] = relationship(back_populates="project")
    pos: Mapped[list["PurchaseOrder"]] = relationship(back_populates="project")

class CostCode(Base):
    __tablename__ = "costcodes"
    id: Mapped[int] = mapped_column(primary_key=True)
    project_id: Mapped[int] = mapped_column(ForeignKey("projects.id"))
    code: Mapped[str] = mapped_column(String(50))
    description: Mapped[str] = mapped_column(String(200))
    project: Mapped[Project] = relationship(back_populates="costcodes")

    __table_args__ = (UniqueConstraint("project_id", "code", name="uq_project_costcode"),)

class ChangeOrder(Base):
    __tablename__ = "change_orders"
    id: Mapped[int] = mapped_column(primary_key=True)
    project_id: Mapped[int] = mapped_column(ForeignKey("projects.id"))
    cor_number: Mapped[str] = mapped_column(String(50))
    title: Mapped[str] = mapped_column(String(200))
    description: Mapped[str] = mapped_column(String(1000))
    type: Mapped[CorType] = mapped_column(Enum(CorType))
    originator: Mapped[str] = mapped_column(String(50))  # RFI/ADDENDUM/EMAIL/OWNER_DIRECTIVE/OTHER
    status: Mapped[CorStatus] = mapped_column(Enum(CorStatus), default=CorStatus.DRAFT)
    estimated_cost: Mapped[float | None] = mapped_column(Numeric(12,2))
    final_cost: Mapped[float | None] = mapped_column(Numeric(12,2))
    existing_costcode_id: Mapped[int | None] = mapped_column(ForeignKey("costcodes.id"), nullable=True)
    new_scope_code: Mapped[str | None] = mapped_column(String(50))
    new_scope_description: Mapped[str | None] = mapped_column(String(200))
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)
    approved_at: Mapped[datetime | None] = mapped_column(DateTime, nullable=True)
    po_created_at: Mapped[datetime | None] = mapped_column(DateTime, nullable=True)

    project: Mapped[Project] = relationship(back_populates="cors")
    existing_costcode: Mapped[CostCode | None] = relationship()
    docs: Mapped[list["ChangeOrderDoc"]] = relationship(back_populates="cor", cascade="all,delete-orphan")
    po: Mapped["PurchaseOrder | None"] = relationship(back_populates="cor", uselist=False)

    __table_args__ = (UniqueConstraint("project_id", "cor_number", name="uq_project_cornumber"),)

class ChangeOrderDoc(Base):
    __tablename__ = "change_order_docs"
    id: Mapped[int] = mapped_column(primary_key=True)
    cor_id: Mapped[int] = mapped_column(ForeignKey("change_orders.id"))
    kind: Mapped[DocKind] = mapped_column(Enum(DocKind))
    filename: Mapped[str] = mapped_column(String(255))
    path: Mapped[str] = mapped_column(String(500))
    cor: Mapped[ChangeOrder] = relationship(back_populates="docs")

class PurchaseOrder(Base):
    __tablename__ = "purchase_orders"
    id: Mapped[int] = mapped_column(primary_key=True)
    project_id: Mapped[int] = mapped_column(ForeignKey("projects.id"))
    po_number: Mapped[str] = mapped_column(String(50))
    cor_id: Mapped[int] = mapped_column(ForeignKey("change_orders.id"), unique=True)
    amount: Mapped[float] = mapped_column(Numeric(12,2))
    costcode_id: Mapped[int] = mapped_column(ForeignKey("costcodes.id"))
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)

    project: Mapped[Project] = relationship(back_populates="pos")
    cor: Mapped[ChangeOrder] = relationship(back_populates="po")

import os, uuid
from config import UPLOAD_DIR

def save_cor_file(project_number: str, cor_number: str, filename: str, file_bytes: bytes) -> str:
    safe_proj = project_number.replace("/", "_")
    safe_cor = cor_number.replace("/", "_")
    folder = os.path.join(UPLOAD_DIR, safe_proj, safe_cor)
    os.makedirs(folder, exist_ok=True)
    unique = f"{uuid.uuid4().hex}_{filename}"
    path = os.path.join(folder, unique)
    with open(path, "wb") as f:
        f.write(file_bytes)
    return path


services/cor_service.py (core validations + state transitions)

from sqlalchemy.orm import Session
from models import Project, CostCode, ChangeOrder, ChangeOrderDoc, DocKind, CorStatus, CorType
from decimal import Decimal
from datetime import datetime

def get_or_create_costcode(db: Session, project: Project, code: str, description: str | None = None) -> CostCode:
    cc = db.query(CostCode).filter_by(project_id=project.id, code=code).one_or_none()
    if cc: return cc
    cc = CostCode(project_id=project.id, code=code, description=description or code)
    db.add(cc); db.flush()
    return cc

def ensure_originator_backup_exists(db: Session, cor: ChangeOrder) -> bool:
    return db.query(ChangeOrderDoc).filter_by(cor_id=cor.id, kind=DocKind.ORIGINATOR_BACKUP).count() > 0

def ensure_approval_backup_exists(db: Session, cor: ChangeOrder) -> bool:
    return db.query(ChangeOrderDoc).filter_by(cor_id=cor.id, kind=DocKind.APPROVAL_BACKUP).count() > 0

def validate_cor_shape(cor: ChangeOrder):
    if cor.type == CorType.BUDGET_ADJUSTMENT and not cor.existing_costcode_id:
        raise ValueError("Existing cost code required for Budget Adjustment COR.")
    if cor.type == CorType.ADDED_SCOPE and not (cor.new_scope_code and cor.new_scope_description):
        raise ValueError("New scope code & description required for Added Scope COR.")

def move_to_submitted(db: Session, cor: ChangeOrder):
    if not ensure_originator_backup_exists(db, cor):
        raise ValueError("Cannot submit: originator backup is required.")
    validate_cor_shape(cor)
    cor.status = CorStatus.SUBMITTED

def move_to_approved(db: Session, cor: ChangeOrder):
    if not ensure_approval_backup_exists(db, cor):
        raise ValueError("Cannot approve: approval backup is required.")
    if cor.final_cost is None:
        raise ValueError("Cannot approve: final_cost must be set.")
    validate_cor_shape(cor)
    cor.status = CorStatus.APPROVED
    cor.approved_at = datetime.utcnow()


services/po_service.py (create PO with gates)

from sqlalchemy.orm import Session
from models import Project, ChangeOrder, PurchaseOrder, CorStatus, CorType
from decimal import Decimal
from datetime import datetime
from .cor_service import get_or_create_costcode

def create_po_from_cor(db: Session, project: Project, cor: ChangeOrder, po_number: str, amount: Decimal, costcode_code: str | None):
    if cor.status != CorStatus.APPROVED:
        raise ValueError("PO creation blocked: COR must be APPROVED.")
    if amount is None:
        raise ValueError("PO amount required.")
    # determine cost code
    if cor.type == CorType.BUDGET_ADJUSTMENT:
        cc_id = cor.existing_costcode_id
    else:
        code = costcode_code or cor.new_scope_code
        desc = cor.new_scope_description or code
        cc = get_or_create_costcode(db, project, code, desc)
        cc_id = cc.id
    po = PurchaseOrder(
        project_id=project.id,
        po_number=po_number,
        cor_id=cor.id,
        amount=amount,
        costcode_id=cc_id
    )
    db.add(po)
    cor.status = CorStatus.PO_CREATED
    cor.po_created_at = datetime.utcnow()
    return po


services/reporting.py

from sqlalchemy.orm import Session
from sqlalchemy import func
from models import Project, ChangeOrder, CorStatus

def cor_summary(db: Session, project_number: str):
    project = db.query(Project).filter_by(number=project_number).one()
    q = db.query(ChangeOrder).filter_by(project_id=project.id)
    total_pending = sum([float(c.estimated_cost or 0) for c in q.filter(ChangeOrder.status.in_([CorStatus.DRAFT, CorStatus.SUBMITTED])).all()])
    total_approved = sum([float(c.final_cost or 0) for c in q.filter(ChangeOrder.status.in_([CorStatus.APPROVED, CorStatus.PO_CREATED])).all()])
    counts = {s.name: db.query(func.count(ChangeOrder.id)).filter_by(project_id=project.id, status=s).scalar() for s in CorStatus}
    return {
        "project": project_number,
        "pending_total": round(total_pending,2),
        "approved_total": round(total_approved,2),
        "counts_by_status": counts
    }

def cor_aging(db: Session, project_number: str):
    from datetime import datetime
    project = db.query(Project).filter_by(number=project_number).one()
    items = []
    for c in db.query(ChangeOrder).filter_by(project_id=project.id).all():
        created = c.created_at
        approved = c.approved_at
        poed = c.po_created_at
        def days(a,b): 
            return ((b - a).days if a and b else None)
        items.append({
            "cor_number": c.cor_number,
            "status": c.status.name,
            "days_draft_to_submitted": None,  # extend if you add timestamps per status
            "days_created_to_approved": days(created, approved),
            "days_approved_to_po": days(approved, poed),
        })
    return {"project": project_number, "aging": items}


routers/api_cor.py (CRUD, uploads, status transitions)

from fastapi import APIRouter, Depends, UploadFile, Form
from sqlalchemy.orm import Session
from db import get_db
from models import Project, CostCode, ChangeOrder, ChangeOrderDoc, DocKind, CorStatus, CorType
from schemas import CORCreate, CORUpdate
from services.files import save_cor_file
from services.cor_service import move_to_submitted, move_to_approved, validate_cor_shape
from decimal import Decimal

router = APIRouter(prefix="/api/cor", tags=["COR"])

def _get_project(db: Session, project_number: str) -> Project:
    p = db.query(Project).filter_by(number=project_number).one_or_none()
    if not p: 
        p = Project(number=project_number, name=project_number)
        db.add(p); db.flush()
    return p

@router.post("")
def create_cor(payload: CORCreate, db: Session = Depends(get_db)):
    project = _get_project(db, payload.project_number)
    cor = ChangeOrder(
        project_id=project.id,
        cor_number=payload.cor_number,
        title=payload.title,
        description=payload.description,
        type=CorType(payload.type),
        originator=payload.originator,
        estimated_cost=payload.estimated_cost
    )
    if payload.existing_costcode_code:
        cc = db.query(CostCode).filter_by(project_id=project.id, code=payload.existing_costcode_code).one_or_none()
        if cc: cor.existing_costcode_id = cc.id
    if payload.type == "ADDED_SCOPE":
        cor.new_scope_code = payload.new_scope_code
        cor.new_scope_description = payload.new_scope_description
    db.add(cor); db.flush()
    return {"id": cor.id, "cor_number": cor.cor_number}

@router.post("/{project_number}/{cor_number}/upload")
async def upload_doc(project_number: str, cor_number: str, kind: DocKind = Form(...), file: UploadFile = Form(...), db: Session = Depends(get_db)):
    project = _get_project(db, project_number)
    cor = db.query(ChangeOrder).filter_by(project_id=project.id, cor_number=cor_number).one()
    bytes_ = await file.read()
    path = save_cor_file(project_number, cor_number, file.filename, bytes_)
    doc = ChangeOrderDoc(cor_id=cor.id, kind=kind, filename=file.filename, path=path)
    db.add(doc); db.flush()
    return {"ok": True, "path": path}

@router.patch("/{project_number}/{cor_number}")
def update_cor(project_number: str, cor_number: str, payload: CORUpdate, db: Session = Depends(get_db)):
    project = _get_project(db, project_number)
    cor = db.query(ChangeOrder).filter_by(project_id=project.id, cor_number=cor_number).one()
    for k,v in payload.model_dump(exclude_unset=True).items():
        if k == "existing_costcode_code" and v:
            cc = db.query(CostCode).filter_by(project_id=project.id, code=v).one_or_none()
            if cc: cor.existing_costcode_id = cc.id
        elif k == "status":
            setattr(cor, "status", CorStatus(v))
        else:
            setattr(cor, k, v)
    db.flush()
    return {"ok": True}

@router.post("/{project_number}/{cor_number}/submit")
def submit_cor(project_number: str, cor_number: str, db: Session = Depends(get_db)):
    project = _get_project(db, project_number)
    cor = db.query(ChangeOrder).filter_by(project_id=project.id, cor_number=cor_number).one()
    move_to_submitted(db, cor)
    return {"status": cor.status.name}

@router.post("/{project_number}/{cor_number}/approve")
def approve_cor(project_number: str, cor_number: str, final_cost: Decimal, db: Session = Depends(get_db)):
    project = _get_project(db, project_number)
    cor = db.query(ChangeOrder).filter_by(project_id=project.id, cor_number=cor_number).one()
    cor.final_cost = final_cost
    move_to_approved(db, cor)
    return {"status": cor.status.name}

app.py

from fastapi import FastAPI
from fastapi.staticfiles import StaticFiles
from fastapi.templating import Jinja2Templates
from sqlalchemy import text
from db import engine, Base
from routers import ui, api_cor, api_po, api_reporting

app = FastAPI(title="AiPM COR Module")
app.mount("/static", StaticFiles(directory="static"), name="static")
templates = Jinja2Templates(directory="templates")
app.state.templates = templates

# Create tables on boot (Alembic can be added later)
Base.metadata.create_all(bind=engine)

app.include_router(ui.router)
app.include_router(api_cor.router)
app.include_router(api_po.router)
app.include_router(api_reporting.router)

@app.get("/health")
def health():
    with engine.connect() as c:
        c.execute(text("select 1"))
    return {"ok": True}
